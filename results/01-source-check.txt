 2559  static u64 calculate_overage(unsigned long usage, unsigned long high)
 2560  {
 2561  	u64 overage;
 2562  
 2563  	if (usage <= high)
 2564  		return 0;

 2566  	/*
 2567  	 * Prevent division by 0 in overage calculation by acting as if
 2568  	 * it was a threshold of 1 page

 2612  					  unsigned int nr_pages,
 2613  					  u64 max_overage)
 2614  {
 2615  	unsigned long penalty_jiffies;
 2616  
 2617  	if (!max_overage)

 2620  	/*
 2621  	 * We use overage compared to memory.high to calculate the number of
 2622  	 * jiffies to sleep (penalty_jiffies). Ideally this value should be
 2623  	 * fairly lenient on small overages, and increasingly harsh when the
 2624  	 * memcg in question makes it clear that it has no intention of stopping
 2625  	 * its crazy behaviour, so we exponentially increase the delay based on
 2626  	 * overage amount.
 2627  	 */
 2628  	penalty_jiffies = max_overage * max_overage * HZ;
 2629  	penalty_jiffies >>= MEMCG_DELAY_PRECISION_SHIFT;
 2630  	penalty_jiffies >>= MEMCG_DELAY_SCALING_SHIFT;

 2634  	 * N-sized allocations are throttled approximately the same as one
 2635  	 * 4N-sized allocation.
 2636  	 *

 2638  	 * larger the current charge patch is than that.
 2639  	 */
